interface LayoutTemplate {

### 3. 布局选择器实现

```typescript
// 根据图片数量推荐布局
function recommendLayouts(imageCount: number): LayoutTemplate[] {
  const layouts: LayoutTemplate[] = getAllLayouts();
  
  return layouts.filter(layout => {
    // 基于列数估算可能的图片数量范围
    const minImages = layout.gridCols;
    const maxImages = layout.gridCols * 4; // 假设最多4行
    
    return imageCount >= minImages && imageCount <= maxImages;
  }).sort((a, b) => {
    // 优先推荐常用的2-4列布局
    if (a.gridCols >= 2 && a.gridCols <= 4) return -1;
    if (b.gridCols >= 2 && b.gridCols <= 4) return 1;
    return 0;
  });
}
```

### 4. 自适应布局渲染

```typescript
function renderLayout(layoutId: number, images: string[]) {
  const layout = getLayoutById(layoutId);
  const cols = layout.gridCols;
  const rows = Math.ceil(images.length / cols);
  
  // 使用 CSS Grid 渲染
  return {
    display: 'grid',
    gridTemplateColumns: `repeat(${cols}, 1fr)`,
    gridTemplateRows: `repeat(${rows}, 1fr)`,
    gap: spacing,
    images: images.slice(0, cols * rows)
  };
}
```
